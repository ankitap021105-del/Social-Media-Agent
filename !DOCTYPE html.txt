/* ── Trending section mini chart init ── */
function initTrendingChart() {
  const ctx = document.getElementById("chart-platform-trending");
  if (!ctx || window._trendPieInit) return;
  window._trendPieInit = true;
  const isDark = document.documentElement.getAttribute("data-theme") === "dark";
  const tc = isDark ? "#e6edf3" : "#1f2328";
  new Chart(ctx, {
    type: "doughnut",
    data: {
      labels: ["Instagram","X (Twitter)","Facebook","LinkedIn"],
      datasets: [{ data: [38,24,21,17], backgroundColor: ["#e1306c","#1da1f2","#1877f2","#0a66c2"], borderWidth: 0 }]
    },
    options: {
      responsive: true, maintainAspectRatio: false,
      plugins: { legend: { position: "right", labels: { color: tc, padding: 10 } } }
    }
  });
}

/* ── Top Posts Table ── */
function renderTopPostsTable() {
  const tb = document.getElementById("top-posts-table");
  if (!tb) return;
  const sorted = [...MOCK_POSTS].sort((a,b) => (b.likes+b.comments+b.shares) - (a.likes+a.comments+a.shares));
  tb.innerHTML = sorted.map((p,i) => `
    <tr>
      <td style="font-weight:700">${i+1}</td>
      <td><span style="color:${platformColor(p.platform)}"><i class="${platformIcon(p.platform)}"></i> ${p.platform}</span></td>
      <td style="max-width:220px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap" title="${p.content}">${p.content.slice(0,55)}…</td>
      <td><strong>${formatNumber(p.likes)}</strong></td>
      <td>${formatNumber(p.comments)}</td>
      <td>${formatNumber(p.shares)}</td>
      <td>${formatNumber(p.reach)}</td>
      <td><span class="sentiment-badge ${p.sentiment}">${sentimentEmoji(p.sentiment)} ${capitalize(p.sentiment)}</span></td>
    </tr>`).join("");
}

/* ── Quick compose helpers ── */
function composePost() {
  const txt = document.getElementById("quick-post-text")?.value?.trim();
  if (!txt) return showToast("Write something first!", "error");
  MOCK_POSTS.unshift({
    id: Date.now(), platform: "Instagram", author: "@my_brand",
    avatar: "https://ui-avatars.com/api/?name=My+Brand&background=3b82d4&color=fff&size=48",
    content: txt, image: null,
    likes: 0, comments: 0, shares: 0, time: "Just now",
    sentiment: "positive", sentimentScore: 0.75,
    hashtags: [], reach: 0
  });
  document.getElementById("quick-post-text").value = "";
  showToast("Post published to feed!", "success");
}

function scheduleQuickPost() {
  const txt = document.getElementById("quick-post-text")?.value?.trim();
  if (!txt) return showToast("Write something first!", "error");
  SCHEDULED_POSTS.unshift({ id: Date.now(), platform: "Instagram", content: txt.slice(0,60), scheduledFor: "Pending…", status: "draft" });
  showToast("Post saved as draft!", "info");
}

function fillFromCaption() {
  const cap = document.getElementById("caption-output")?.textContent;
  const qp  = document.getElementById("quick-post-text");
  if (!cap || cap.includes("appear here") || !qp) return showToast("Generate a caption first!", "error");
  qp.value = cap;
  showToast("Caption copied to composer!", "success");
}

  if (page === "trending") { setTimeout(initTrendingChart, 100); }
  if (page === "analytics") { setTimeout(renderTopPostsTable, 200); }
};
